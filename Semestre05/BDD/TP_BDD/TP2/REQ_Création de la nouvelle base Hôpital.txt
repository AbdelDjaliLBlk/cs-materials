-- 1. Afficher les informations sur les tablespaces
SELECT TABLESPACE_NAME, FILE_NAME, BYTES/1024/1024 AS SIZE_MB, STATUS 
FROM DBA_DATA_FILES;

-- 2. Afficher la tablespace par défaut
SELECT PROPERTY_NAME, PROPERTY_VALUE 
FROM DATABASE_PROPERTIES
WHERE PROPERTY_NAME = 'DEFAULT_PERMANENT_TABLESPACE';

-- 3. Créer les tablespaces demandés
CREATE TABLESPACE BDDING_TBS
DATAFILE 'C:\app\oracle\oradata\BDDING\HospitalisationFile01.dbf'
SIZE 50M AUTOEXTEND ON NEXT 10M MAXSIZE 100M;

CREATE TEMPORARY TABLESPACE BDDING_TEMP_TBS
TEMPFILE 'C:\app\oracle\oradata\BDDING\TempFile01.dbf'
SIZE 50M;

-- 4. Créer l’utilisateur USERBDDING
CREATE USER USERBDDING IDENTIFIED BY userbdd123
DEFAULT TABLESPACE BDDING_TBS
QUOTA 100M ON USERS;

-- 5. Accorder les privilèges nécessaires
GRANT CONNECT, RESOURCE TO USERBDDING;
